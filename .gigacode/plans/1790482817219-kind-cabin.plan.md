## Шаг 1: Извлечение USER_ID в shared utility

**Файл:** `js/sadhana-utils.js` (создать новый)

```javascript
// Получение USER_ID из сессии
export async function getCurrentUserId() {
    const cached = localStorage.getItem('user_id');
    if (cached) return cached;
    
    try {
        const res = await fetch('/api/check_auth.php');
        if (res.ok) {
            const data = await res.json();
            const userId = data.user_id || data.id;
            if (userId) {
                localStorage.setItem('user_id', userId);
                return userId;
            }
        }
    } catch (e) {
        console.warn('Could not fetch user ID:', e);
    }
    
    // Fallback: try userInfo localStorage
    try {
        const info = JSON.parse(localStorage.getItem('userInfo') || '{}');
        return info.id || info.user_id || null;
    } catch {
        return null;
    }
}

// API helper с retry
export async function apiFetch(url, options = {}, retries = 2) {
    for (let i = 0; i <= retries; i++) {
        try {
            const response = await fetch(url, options);
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}`);
            }
            return await response.json();
        } catch (error) {
            if (i === retries) throw error;
            await new Promise(r => setTimeout(r, 500 * (i + 1)));
        }
    }
}
```

**JavaScript:**
```javascript
async function loadLeaderboard() {
    try {
        const response = await fetch(`${API_BASE}/api/sadhana/leaderboard?limit=10`);
        if (response.ok) {
            const users = await response.json();
            renderLeaderboard(users);
        }
    } catch (error) {
        console.log('Leaderboard load failed:', error);
    }
}

function renderLeaderboard(users) {
    const container = document.getElementById('leaderboardList');
    container.innerHTML = users.map((user, i) => `
        <div class="leaderboard-item">
            <span class="rank">#${i + 1}</span>
            <span class="name">${escapeHtml(user.name)}</span>
            <span class="progress">${user.completionPercent}%</span>
        </div>
    `).join('');
}

function validateInput(value, min = 0) {
    const num = Number(value);
    if (isNaN(num) || num < min) return min;
    return num;
}

// Retry в saveDailyLog
async function saveWithRetry(url, body, maxRetries = 2) {
    for (let i = 0; i <= maxRetries; i++) {
        try {
            const res = await fetch(url, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json', 'X-User-Id': USER_ID },
                body: JSON.stringify(body),
            });
            if (!res.ok) throw new Error(`HTTP ${res.status}`);
            return true;
        } catch (e) {
            if (i === maxRetries) throw e;
            await new Promise(r => setTimeout(r, 500 * (i + 1)));
        }
    }
}
```
